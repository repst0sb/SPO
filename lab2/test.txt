func processMetricsData(inputVal: Int, numbers: [Int]) -> Int {
    var total = 0;
    
    if inputVal > 0 {
        total += inputVal;
    } else {
        total -= inputVal;
    }
    
    for number in numbers {
        if number % 2 == 0 {
            total += number;
        } else {
            total -= number;
        }
    }
    
    var counter = 2;
    while counter > 0 {
        total *= 10;
        counter -= 1;
    }
    
    repeat {
        total -= 1;
    } while total > 1000;
    
    if inputVal > 5 {
        switch total {
        case 10:
            total = 100;
        case 20:
            total = 200;
        case 30:
            total = 300;
        case 40:
            total = 400;
        default:
            total = 0;
        }
    }
    
    return total;
}
